# 用户肖像系统设计方案

## 1. 核心挑战

机器人服务多用户场景下的用户肖像系统需要解决：
1. **多用户隔离** - 每个用户有独立的肖像
2. **持续学习** - 随对话增量更新，而非每次重建
3. **偏好演化** - 追踪用户偏好随时间的变化
4. **高效检索** - 不能每次都全量扫描
5. **一致性** - 跨会话保持用户认知

```sql
CREATE TABLE user_portraits (

    portrait_summary TEXT,          -- LLM 生成的 2-3 句描述
    portrait_embedding BLOB,        -- 肖像的向量表示

    dislikes JSON,

    habits JSON,
    skills JSON,
    goals JSON,

    -- 元数据
    memory_count INTEGER DEFAULT 0,
    confidence REAL DEFAULT 0.0,

    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_accessed TIMESTAMP,

    -- 版本控制
    version INTEGER DEFAULT 1,
    update_count INTEGER DEFAULT 0
);
```

### 3.2 偏好演化表 (preference_evolution)

```sql
CREATE TABLE preference_evolution (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id TEXT,

    -- 偏好变化
    preference_type TEXT,           -- 'likes', 'dislikes', etc.
    old_value TEXT,                 -- 之前的值
    new_value TEXT,                 -- 新的值
    change_type TEXT,               -- 'added', 'removed', 'modified'

    -- 变化原因
    trigger_memory_id TEXT,         -- 触发变化的记忆
    trigger_statement TEXT,         -- 原始陈述

    -- 时间
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (user_id) REFERENCES user_portraits(user_id)
);
```

## 4. 增量更新机制

### 4.1 偏好提取 → 增量更新流程

```python
class UserPortraitManager:
    """用户肖像管理器 - 支持增量更新"""

    def __init__(self, db_path: str, cache_ttl: int = 3600):
        self.db_path = db_path
        self.cache = {}  # user_id → Portrait (内存缓存)
        self.cache_ttl = cache_ttl

    async def on_new_preference(self, user_id: str, preference: Dict) -> None:
        """新偏好进入时的增量更新"""

        # 1. 获取当前肖像 (从缓存或数据库)
        portrait = await self.get_portrait(user_id)

        # 2. 增量更新
        pref_type = preference.get('preference_type', 'other')
        pref_value = preference.get('preference_value')

        if pref_type in portrait.structured:
            current_list = portrait.structured[pref_type]

            # 检查是否已存在（去重）
            if pref_value not in current_list:
                # 添加新偏好
                current_list.append(pref_value)

                # 记录偏好演化
                await self._record_evolution(
                    user_id=user_id,
                    pref_type=pref_type,
                    old_value=None,
                    new_value=pref_value,
                    change_type='added',
                    trigger=preference.get('original_statement')
                )

        # 3. 判断是否需要重新合成肖像描述
        if portrait.update_count % 5 == 0:  # 每 5 次更新重新合成
            portrait.summary = await self._regenerate_summary(portrait)

        # 4. 更新置信度
        portrait.confidence = self._calculate_confidence(portrait)

        # 5. 保存
        await self._save_portrait(portrait)

        # 6. 更新缓存
        self.cache[user_id] = portrait
```

### 4.2 置信度计算

```python
def _calculate_confidence(self, portrait: UserPortrait) -> float:
    """
    置信度计算考虑：
    1. 偏好数量
    2. 偏好多样性（覆盖多少类别）
    3. 时间因素（最近是否有更新）
    4. 一致性（是否有矛盾的偏好）
    """

    # 数量分数 (0-0.4)
    total_prefs = sum(len(v) for v in portrait.structured.values())
    quantity_score = min(0.4, total_prefs / 50)  # 50个偏好达到满分

    # 多样性分数 (0-0.3)
    categories_with_data = sum(1 for v in portrait.structured.values() if len(v) > 0)
    diversity_score = min(0.3, categories_with_data / 7 * 0.3)  # 7个类别

    # 时间分数 (0-0.2)
    days_since_update = (datetime.now() - portrait.updated_at).days
    recency_score = max(0, 0.2 - days_since_update * 0.02)  # 10天后降为0

    # 一致性分数 (0-0.1)
    # 检查 likes 和 dislikes 是否有冲突
    conflicts = set(portrait.structured['likes']) & set(portrait.structured['dislikes'])
    consistency_score = 0.1 if len(conflicts) == 0 else 0.0

    return quantity_score + diversity_score + recency_score + consistency_score
```

## 5. 偏好演化追踪

### 5.1 变化检测

```python
async def detect_preference_change(
    self,
    user_id: str,
    new_statement: str
) -> Optional[PreferenceChange]:
    """
    检测用户偏好是否发生变化

    例如：
    - 之前: "I love coffee"
    - 现在: "I've stopped drinking coffee"

    返回: PreferenceChange(type='likes', old='coffee', new=None, change='removed')
    """

    # 使用 LLM 分析是否是偏好变化
    portrait = await self.get_portrait(user_id)

    prompt = f"""Analyze if this statement indicates a preference CHANGE from the user's known preferences.

Known user preferences:
- Likes: {portrait.structured.get('likes', [])}
- Dislikes: {portrait.structured.get('dislikes', [])}
- Interests: {portrait.structured.get('interests', [])}

New statement: "{new_statement}"

If this is a preference CHANGE (not just a new preference), output:
{{
    "is_change": true,
    "type": "likes|dislikes|interests|...",
    "old_value": "the previous preference",
    "new_value": "the new preference or null if removed",
    "change_type": "modified|removed"
}}

If this is NOT a change, output:
{{"is_change": false}}
"""

    result = await self.llm.analyze(prompt)
    if result.get('is_change'):
        return PreferenceChange(**result)
    return None
```

## 6. 多用户隔离策略

### 6.1 user_id 生成策略

```python
def generate_user_id(context: Dict) -> str:
    """
    根据上下文生成用户标识

    策略优先级:
    1. 显式传入的 user_id
    2. 会话 session_id + 用户名
    3. 设备指纹 (如有)
    4. 默认 'anonymous_{session_id}'
    """

    if context.get('user_id'):
        return context['user_id']

    session_id = context.get('session_id', 'unknown')
    username = context.get('username')

    if username:
        return f"{username}_{session_id[:8]}"

    device_fp = context.get('device_fingerprint')
    if device_fp:
        return f"device_{device_fp[:16]}"

    return f"anonymous_{session_id[:8]}"
```

### 6.2 用户隔离检索

```python
async def retrieve_with_portrait(
    self,
    query: str,
    user_id: str
) -> Dict:
    """
    检索时注入用户肖像上下文
    """

    # 1. 获取用户肖像
    portrait = await self.portrait_manager.get_portrait(user_id)

    # 2. 检索记忆 (带 user_id 过滤)

        'retrieved_memories': memories,
        'user_preferences': portrait.structured
    }

    return context
```

## 7. 缓存策略

### 7.1 多级缓存

```
┌─────────────────────────────────────────┐
│           L1: 进程内存缓存              │
│  - 最近 100 个活跃用户的肖像            │
│  - TTL: 10 分钟                         │
│  - 命中率目标: >80%                     │
└─────────────────────────────────────────┘
                    ↓ miss
┌─────────────────────────────────────────┐
│           L2: Redis 缓存                │
│  - 所有用户的肖像                       │
│  - TTL: 1 小时                          │
│  - 支持分布式部署                       │
└─────────────────────────────────────────┘
                    ↓ miss
┌─────────────────────────────────────────┐
│           L3: SQLite/PostgreSQL         │
│  - 持久化存储                           │
│  - 偏好演化历史                         │
└─────────────────────────────────────────┘
```

### 7.2 缓存失效策略

```python
async def invalidate_cache(self, user_id: str, reason: str):
    """
    缓存失效触发条件:
    1. 用户偏好更新时
    2. 检测到偏好变化时
    3. 用户明确要求重置时
    4. 定期刷新 (每小时)
    """

    # 删除 L1 缓存
    if user_id in self.l1_cache:
        del self.l1_cache[user_id]

    # 删除 L2 缓存
    await self.redis.delete(f"portrait:{user_id}")

    logger.info(f"Cache invalidated for {user_id}: {reason}")
```
